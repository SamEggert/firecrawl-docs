```js Node.js
// npm install @mendable/firecrawl-js

import { Firecrawl } from '@mendable/firecrawl-js';

const firecrawl = new Firecrawl({ apiKey: 'YOUR_API_KEY' });

// Función para raspar con lógica de reintentos
async function scrapeWithRetry(url) {
  try {
    // Primer intento con el proxy predeterminado
    const content = await firecrawl.scrape(url);
    
    // Comprobar si obtuvimos un código de estado de error
    const statusCode = content?.metadata?.statusCode;
    if ([401, 403, 500].includes(statusCode)) {
      console.log(`Se obtuvo el código de estado ${statusCode}, reintentando con proxy stealth`);
      // Reintentar con proxy stealth
      return await firecrawl.scrape(url, {
        proxy: 'stealth'
      });
    }
    
    return content;
  } catch (error) {
    console.error(`Error: ${error.message}`);
    // Reintentar con proxy stealth en caso de excepción
    try {
      return await firecrawl.scrape(url, {
        proxy: 'stealth'
      });
    } catch (retryError) {
      console.error(`El proxy stealth también falló: ${retryError.message}`);
      throw retryError;
    }
  }
}

// Ejemplo de uso
const content = await scrapeWithRetry('https://example.com');
console.log(content.markdown);
```
