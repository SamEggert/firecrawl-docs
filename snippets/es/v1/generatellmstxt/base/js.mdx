```js Node
import FirecrawlApp from "firecrawl";

// Inicializa el cliente
const firecrawl = new FirecrawlApp({ apiKey: "your_api_key" });

// Define los parámetros de generación
const params = {
  maxUrls: 2, // Máximo de URL a analizar
  showFullText: true, // Incluir el texto completo en los resultados
};

// Genera LLMs.txt con sondeo
const results = await firecrawl.generateLLMsText("https://example.com", params);

// Accede a los resultados de la generación
if (results.success) {
  console.log(`Estado: ${results.status}`);
  console.log(`Datos generados:`, results.data);
} else {
  console.error(`Error: ${results.error || "Error desconocido"}`);
}
```
